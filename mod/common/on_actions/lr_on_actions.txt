on_actions = {
	on_startup = {
		effect = {
			# The scenario is written for the 1936 bookmark only.
			if = {
				limit = { date < 1936.2.1 }

				USA = {
					if = {
						limit = { is_in_faction = yes }
						leave_faction = yes
					}
				}

				# on_startup has no scope, and a scripted effect needs one.
				log = "LR: startup begins"
				FRA = { lr_setup_world = yes }
				# Who owned each state when the game began, for the peace treaties.
				every_state = {
					OWNER = {
						PREV = { set_variable = { LR_start_owner = PREV } }
					}
				}
				every_state = {
					limit = { is_owned_by = FRA }
					set_state_flag = LR_start_FRA
				}
				every_state = {
					limit = { is_owned_by = SOV }
					set_state_flag = LR_start_SOV
				}
				807 = { log = "LR: Danzig owner after setup is [THIS.Owner.GetTag]" }
				159 = { log = "LR: Lombardy owner after setup is [THIS.Owner.GetTag]" }
				log = "LR: startup finished"

				# France: this world's defeated power.
				FRA = {
					remove_ideas = FRA_victors_of_wwi
					remove_ideas = FRA_disjointed_government
					remove_ideas = FRA_protected_by_the_maginot_line
					# Every vanilla French spirit belongs to another history.
					remove_ideas = FRA_full_employment
					remove_ideas = FRA_political_violence
					remove_ideas = FRA_inefficient_economy_2
					remove_ideas = FRA_matignon_agreements
					remove_ideas = MTG_naval_treaty_adherent
					remove_ideas = MTG_naval_treaty_adherent_reduced
					lr_fra_spirits_init = yes
					add_stability = -0.2
					lr_fra_sync_industry = yes
					29 = { set_demilitarized_zone = yes }
					1077 = { set_demilitarized_zone = yes }
					1131 = { set_demilitarized_zone = yes }
					set_variable = { LR_rail_lines = 0 }
					set_variable = { LR_road_works = 0 }
					set_variable = { LR_funds = 50 }
					set_variable = { LR_pk_hands_left = 3 }
					lr_pk_init = yes
					promote_character = FRA_louis_xx
					# The Algerian revolt, in early 1936.
					country_event = { id = lr_novel.101 days = 14 }
				}

				SOV = {
					promote_character = LR_SOV_kerensky
				}

				JAP = {
					add_ideas = LR_JAP_han_autonomy
				}

				# Spain, China, Spanish America, Washington, Japan and Italy, on fixed dates.
				# on_startup has no scope, and a scripted effect needs one.
				FRA = { lr_world_startup = yes }
			}
		}
	}
	# ROOT declares war on FROM.
	# The second world war's super event is checked here and each month.
	on_declare_war = {
		effect = {
			lr_league_russia_joins = yes
			lr_check_world_war = yes
		}
	}
	# Regional opposition and economy projects advance each month.
	on_monthly_FRA = {
		effect = {
			lr_fra_monthly = yes
			lr_fra_power_monthly = yes
			lr_check_world_war = yes
		}
	}
	# Runs the moment a country capitulates. ROOT lost, FROM won. France's own wars go on to the peace conference;
	# here it only records who fell.
	on_capitulation_immediate = {
		effect = {
			lr_world_on_capitulation = yes
			if = {
				limit = {
					ROOT = { tag = GER }
					NOT = { has_global_flag = LR_settled_germany }
					FROM = { OR = { tag = FRA is_in_faction_with = FRA is_subject_of = FRA tag = SOV tag = YUG tag = ROM tag = GRE } }
				}
				FRA = {
					set_global_flag = LR_settled_germany
					log = "LR: GER capitulated to [FROM.GetTag], settling"
					set_country_flag = LR_germany_defeated
					# North Germany's colonies go to a government in exile at Douala, not to the conference table.
					lr_german_exile = yes
					news_event = { id = lr_news.10 days = 1 }
				}
			}
			if = {
				limit = {
					ROOT = { tag = AUS }
					NOT = { has_global_flag = LR_settled_austria }
					FROM = { OR = { tag = FRA is_in_faction_with = FRA is_subject_of = FRA tag = SOV tag = YUG tag = ROM tag = GRE } }
				}
				FRA = {
					set_global_flag = LR_settled_austria
					log = "LR: AUS capitulated to [FROM.GetTag], settling"
					set_country_flag = LR_austria_defeated
					news_event = { id = lr_news.11 days = 1 }
				}
			}
			if = {
				limit = {
					ROOT = { tag = ENG }
					NOT = { has_global_flag = LR_settled_britain }
					FROM = { OR = { tag = FRA is_in_faction_with = FRA is_subject_of = FRA } }
				}
				FRA = {
					set_global_flag = LR_settled_britain
					log = "LR: ENG capitulated to [FROM.GetTag], settling"
					news_event = { id = lr_news.12 days = 1 }
				}
			}
			if = {
				limit = {
					ROOT = { tag = CAN }
					has_global_flag = LR_crown_in_canada
					NOT = { has_global_flag = LR_settled_canada }
					FROM = { OR = { tag = FRA is_in_faction_with = FRA is_subject_of = FRA } }
				}
				FRA = {
					set_global_flag = LR_settled_canada
					log = "LR: CAN capitulated to [FROM.GetTag], settling"
					clr_global_flag = LR_crown_in_canada
					set_global_flag = LR_crown_in_australia
					news_event = { id = lr_news.13 days = 1 }
				}
			}
			# A fallen country makes its own peace, with its subjects, while its faction fights on.
			lr_separate_peace = yes
			# Every war ends at the vanilla peace conference; the land is then settled by region decisions.
		}
	}
}
